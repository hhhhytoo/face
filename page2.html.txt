        // ==================== EYE TRACKING ====================
        const face = document.getElementById('face');
        const leftPupil = document.getElementById('pupil-left');
        const rightPupil = document.getElementById('pupil-right');
        const maxOffset = 6;

        function movePupils(e) {
            const leftEye = document.querySelector('.eye.left');
            const rightEye = document.querySelector('.eye.right');
            if (!leftEye || !rightEye) return;

            const leftRect = leftEye.getBoundingClientRect();
            const rightRect = rightEye.getBoundingClientRect();

            const leftCenter = {
                x: leftRect.left + leftRect.width / 2,
                y: leftRect.top + leftRect.height / 2
            };
            const rightCenter = {
                x: rightRect.left + rightRect.width / 2,
                y: rightRect.top + rightRect.height / 2
            };

            const getOffset = (center) => {
                let dx = e.clientX - center.x;
                let dy = e.clientY - center.y;
                const dist = Math.sqrt(dx * dx + dy * dy);
                if (dist === 0) return { x: 0, y: 0 };
                const ratio = Math.min(maxOffset / dist, 1);
                return { x: dx * ratio, y: dy * ratio };
            };

            const leftOff = getOffset(leftCenter);
            const rightOff = getOffset(rightCenter);

            leftPupil.style.transform = `translate(calc(-50% + ${leftOff.x}px), calc(-50% + ${leftOff.y}px))`;
            rightPupil.style.transform = `translate(calc(-50% + ${rightOff.x}px), calc(-50% + ${rightOff.y}px))`;
        }

        document.addEventListener('mousemove', movePupils);

        // ==================== MOUTH MORPH ====================
        const mouthPath = document.getElementById('mouth-path');
        const smilePath = "M 10,15 Q 30,25 50,15";
        let mouthSmiled = false;
        let animFrame = null;
        let startTime = null;
        const morphDuration = 1000;

        function animateMorph(targetPath) {
            if (animFrame) cancelAnimationFrame(animFrame);
            const startPath = mouthPath.getAttribute('d');
            const startPoints = parsePath(startPath);
            const endPoints = parsePath(targetPath);
            startTime = performance.now();

            function step(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / morphDuration, 1);
                const eased = 0.5 - 0.5 * Math.cos(Math.PI * t);
                const current = interpolatePath(startPoints, endPoints, eased);
                mouthPath.setAttribute('d', current);
                if (t < 1) {
                    animFrame = requestAnimationFrame(step);
                } else {
                    animFrame = null;
                }
            }
            animFrame = requestAnimationFrame(step);
        }

        // Path parsing helpers
        function parsePath(d) {
            const parts = d.split(' ');
            const cmd1 = parts[0];
            const x1 = parseFloat(parts[1].split(',')[0]);
            const y1 = parseFloat(parts[1].split(',')[1]);
            const cmd2 = parts[2];
            if (cmd2 === 'L') {
                const x2 = parseFloat(parts[3].split(',')[0]);
                const y2 = parseFloat(parts[3].split(',')[1]);
                return { type: 'L', start: { x: x1, y: y1 }, end: { x: x2, y: y2 } };
            } else if (cmd2 === 'Q') {
                const cx = parseFloat(parts[3].split(',')[0]);
                const cy = parseFloat(parts[3].split(',')[1]);
                const x2 = parseFloat(parts[4].split(',')[0]);
                const y2 = parseFloat(parts[4].split(',')[1]);
                return { type: 'Q', start: { x: x1, y: y1 }, control: { x: cx, y: cy }, end: { x: x2, y: y2 } };
            }
            return null;
        }

        function interpolatePath(start, end, t) {
            if (start.type === 'L' && end.type === 'Q') {
                const mid = {
                    x: (start.start.x + start.end.x) / 2,
                    y: (start.start.y + start.end.y) / 2
                };
                const startAsQ = {
                    type: 'Q',
                    start: start.start,
                    control: mid,
                    end: start.end
                };
                return interpolateQ(startAsQ, end, t);
            } else if (start.type === 'Q' && end.type === 'L') {
                const mid = {
                    x: (end.start.x + end.end.x) / 2,
                    y: (end.start.y + end.end.y) / 2
                };
                const endAsQ = {
                    type: 'Q',
                    start: end.start,
                    control: mid,
                    end: end.end
                };
                return interpolateQ(start, endAsQ, t);
            } else if (start.type === 'Q' && end.type === 'Q') {
                return interpolateQ(start, end, t);
            } else {
                const x = start.start.x + (end.start.x - start.start.x) * t;
                const y = start.start.y + (end.start.y - start.start.y) * t;
                const x2 = start.end.x + (end.end.x - start.end.x) * t;
                const y2 = start.end.y + (end.end.y - start.end.y) * t;
                return `M ${x},${y} L ${x2},${y2}`;
            }
        }

        function interpolateQ(start, end, t) {
            const sx = start.start.x + (end.start.x - start.start.x) * t;
            const sy = start.start.y + (end.start.y - start.start.y) * t;
            const cx = start.control.x + (end.control.x - start.control.x) * t;
            const cy = start.control.y + (end.control.y - start.control.y) * t;
            const ex = start.end.x + (end.end.x - start.end.x) * t;
            const ey = start.end.y + (end.end.y - start.end.y) * t;
            return `M ${sx},${sy} Q ${cx},${cy} ${ex},${ey}`;
        }

        // ==================== BUBBLE MANAGEMENT ====================
        let bubblesVisible = false;
        let animationTimer = null;

        function showBubbles() {
            const bubbles = document.querySelectorAll('.bubble');
            if (animationTimer) clearTimeout(animationTimer);
            bubbles.forEach(b => {
                b.classList.remove('tl', 'tr', 'bl', 'br', 'floating');
            });
            bubbles.forEach(b => b.classList.add('visible'));

            setTimeout(() => {
                document.getElementById('bubble-tl').classList.add('tl');
                document.getElementById('bubble-tr').classList.add('tr');
                document.getElementById('bubble-bl').classList.add('bl');
                document.getElementById('bubble-br').classList.add('br');
            }, 10);

            setTimeout(() => {
                bubbles.forEach(b => b.classList.add('floating'));
            }, 1100);
        }

        function hideBubbles() {
            const bubbles = document.querySelectorAll('.bubble');
            bubbles.forEach(b => {
                b.classList.remove('floating');
            });
            bubbles.forEach(b => {
                b.classList.remove('tl', 'tr', 'bl', 'br');
            });
            animationTimer = setTimeout(() => {
                bubbles.forEach(b => b.classList.remove('visible'));
            }, 1000);
        }

        // ==================== SCROLL TO PARAGRAPH ====================
        function setupBubbleClicks() {
            const bubbleTL = document.getElementById('bubble-tl');
            const bubbleTR = document.getElementById('bubble-tr');
            const bubbleBL = document.getElementById('bubble-bl');
            const bubbleBR = document.getElementById('bubble-br');

            bubbleTL.addEventListener('click', () => {
                document.getElementById('workout-text').scrollIntoView({ behavior: 'smooth', block: 'center' });
            });
            bubbleTR.addEventListener('click', () => {
                document.getElementById('sleep-text').scrollIntoView({ behavior: 'smooth', block: 'center' });
            });
            bubbleBL.addEventListener('click', () => {
                document.getElementById('food-text').scrollIntoView({ behavior: 'smooth', block: 'center' });
            });
            bubbleBR.addEventListener('click', () => {
                document.getElementById('protein-text').scrollIntoView({ behavior: 'smooth', block: 'center' });
            });
        }

        // ==================== FACE CLICK HANDLER ====================
        face.addEventListener('click', function() {
            if (!mouthSmiled) {
                animateMorph(smilePath);
                mouthSmiled = true;
            }

            if (!bubblesVisible) {
                showBubbles();
                bubblesVisible = true;
            } else {
                hideBubbles();
                bubblesVisible = false;
            }
        });

        // Initialize bubble click listeners
        setupBubbleClicks();

        window.addEventListener('beforeunload', function() {
            if (animationTimer) clearTimeout(animationTimer);
        });
    </script>
