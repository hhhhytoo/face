    <script>
        let bubbleGenerated = false;

        function continueToPage2() {
            if (!bubbleGenerated) return;
            window.location.href = 'page2.html'; // Opens page2
        }

        function generateBubble() {
            const generateBtn = document.getElementById('generate-btn');
            if (generateBtn.disabled) return;

            const usernameInput = document.getElementById('username');
            let username = usernameInput.value.trim();
            if (username === '') {
                username = 'Guest';
            }
            createFloatingBubble(username);

            generateBtn.disabled = true;
            bubbleGenerated = true;
            document.getElementById('continue-btn').disabled = false;
        }

        function createFloatingBubble(username) {
            const container = document.getElementById('black-bubbles');
            const face = document.getElementById('face');
            const faceRect = face.getBoundingClientRect();
            const containerRect = container.getBoundingClientRect();

            const faceCenterX = (faceRect.left + faceRect.width / 2 - containerRect.left) / containerRect.width * 100;
            const faceCenterY = (faceRect.top + faceRect.height / 2 - containerRect.top) / containerRect.height * 100;

            const angle = Math.random() * 2 * Math.PI;
            const radius = 5;
            const startX = faceCenterX + Math.cos(angle) * radius;
            const startY = faceCenterY + Math.sin(angle) * radius;

            const targetX = 80;
            const existingBubbles = document.querySelectorAll('.black-bubble');
            let targetY = 20;
            if (existingBubbles.length > 0) {
                const usedYs = [];
                existingBubbles.forEach(b => {
                    const y = parseFloat(b.dataset.targetY) || 20;
                    usedYs.push(y);
                });
                for (let y = 10; y <= 90; y += 10) {
                    if (!usedYs.some(used => Math.abs(used - y) < 8)) {
                        targetY = y;
                        break;
                    }
                }
                if (targetY === 20) targetY = 20 + (existingBubbles.length * 8) % 80;
            }

            const bubble = document.createElement('div');
            bubble.className = 'black-bubble';
            bubble.textContent = username;
            bubble.style.left = startX + '%';
            bubble.style.top = startY + '%';
            bubble.dataset.targetX = targetX;
            bubble.dataset.targetY = targetY;
            container.appendChild(bubble);

            setTimeout(() => {
                bubble.style.left = targetX + '%';
                bubble.style.top = targetY + '%';
            }, 10);
        }

        // Wrap characters for hover effect
        function wrapCharacters() {
            const elements = document.querySelectorAll('#white-content h1, #white-content p');
            elements.forEach(el => {
                const text = el.textContent;
                const wrapped = text.split('').map(char => {
                    if (char === ' ') return ' ';
                    return `<span class="char">${char}</span>`;
                }).join('');
                el.innerHTML = wrapped;
            });
        }

        // Entrance animations
        window.addEventListener('load', function() {
            wrapCharacters();

            setTimeout(() => {
                document.getElementById('eye-left').classList.add('open');
                document.getElementById('eye-right').classList.add('open');

            }, 3500);

        // Mouth morph on username click
        const mouthPath = document.getElementById('mouth-path');
        const smilePath = "M 10,15 Q 30,25 50,15";

        let animFrame = null;
        let startTime = null;
        const duration = 1000;

        function animateMorph(targetPath) {
            if (animFrame) cancelAnimationFrame(animFrame);
            const startPath = mouthPath.getAttribute('d');
            const startPoints = parsePath(startPath);
            const endPoints = parsePath(targetPath);
            startTime = performance.now();

            function step(now) {
                const elapsed = now - startTime;
                const t = Math.min(elapsed / duration, 1);
                const eased = 0.5 - 0.5 * Math.cos(Math.PI * t);
                const currentPath = interpolatePath(startPoints, endPoints, eased);
                mouthPath.setAttribute('d', currentPath);
                if (t < 1) {
                    animFrame = requestAnimationFrame(step);
                } else {
                    animFrame = null;
                }
            }
            animFrame = requestAnimationFrame(step);
        }

        function parsePath(d) {
            const parts = d.split(' ');
            const cmd1 = parts[0];
            const x1 = parseFloat(parts[1].split(',')[0]);
            const y1 = parseFloat(parts[1].split(',')[1]);
            const cmd2 = parts[2];
            if (cmd2 === 'L') {
                const x2 = parseFloat(parts[3].split(',')[0]);
                const y2 = parseFloat(parts[3].split(',')[1]);
                return { type: 'L', start: { x: x1, y: y1 }, end: { x: x2, y: y2 } };
            } else if (cmd2 === 'Q') {
                const cx = parseFloat(parts[3].split(',')[0]);
                const cy = parseFloat(parts[3].split(',')[1]);
                const x2 = parseFloat(parts[4].split(',')[0]);
                const y2 = parseFloat(parts[4].split(',')[1]);
                return { type: 'Q', start: { x: x1, y: y1 }, control: { x: cx, y: cy }, end: { x: x2, y: y2 } };
            }
            return null;
        }

        function interpolatePath(start, end, t) {
            if (start.type === 'L' && end.type === 'Q') {
                const mid = {
                    x: (start.start.x + start.end.x) / 2,
                    y: (start.start.y + start.end.y) / 2
                };
                const startAsQ = {
                    type: 'Q',
                    start: start.start,
                    control: mid,
                    end: start.end
                };
                return interpolateQ(startAsQ, end, t);
            } else if (start.type === 'Q' && end.type === 'L') {
                const mid = {
                    x: (end.start.x + end.end.x) / 2,
                    y: (end.start.y + end.end.y) / 2
                };
                const endAsQ = {
                    type: 'Q',
                    start: end.start,
                    control: mid,
                    end: end.end
                };
                return interpolateQ(start, endAsQ, t);
            } else if (start.type === 'Q' && end.type === 'Q') {
                return interpolateQ(start, end, t);
            } else {
                const x = start.start.x + (end.start.x - start.start.x) * t;
                const y = start.start.y + (end.start.y - start.start.y) * t;
                const x2 = start.end.x + (end.end.x - start.end.x) * t;
                const y2 = start.end.y + (end.end.y - start.end.y) * t;
                return `M ${x},${y} L ${x2},${y2}`;
            }
        }

        function interpolateQ(start, end, t) {
            const sx = start.start.x + (end.start.x - start.start.x) * t;
            const sy = start.start.y + (end.start.y - start.start.y) * t;
            const cx = start.control.x + (end.control.x - start.control.x) * t;
            const cy = start.control.y + (end.control.y - start.control.y) * t;
            const ex = start.end.x + (end.end.x - start.end.x) * t;
            const ey = start.end.y + (end.end.y - start.end.y) * t;
            return `M ${sx},${sy} Q ${cx},${cy} ${ex},${ey}`;
        }

        document.getElementById('username').addEventListener('click', function() {
            animateMorph(smilePath);
        });

        // Eye tracking
        const face = document.getElementById('face');
        const leftPupil = document.getElementById('pupil-left');
        const rightPupil = document.getElementById('pupil-right');
        const maxOffset = 6;

        function movePupils(event) {
            const leftEye = document.querySelector('.eye.left');
            const rightEye = document.querySelector('.eye.right');
            if (!leftEye || !rightEye) return;

            const leftEyeRect = leftEye.getBoundingClientRect();
            const rightEyeRect = rightEye.getBoundingClientRect();

            const leftEyeCenter = {
                x: leftEyeRect.left + leftEyeRect.width / 2,
                y: leftEyeRect.top + leftEyeRect.height / 2
            };
            const rightEyeCenter = {
                x: rightEyeRect.left + rightEyeRect.width / 2,
                y: rightEyeRect.top + rightEyeRect.height / 2
            };

            function getOffset(eyeCenter) {
                let dx = event.clientX - eyeCenter.x;
                let dy = event.clientY - eyeCenter.y;
                const distance = Math.sqrt(dx * dx + dy * dy);
                if (distance === 0) return { x: 0, y: 0 };
                const ratio = Math.min(maxOffset / distance, 1);
                return { x: dx * ratio, y: dy * ratio };
            }

            const leftOffset = getOffset(leftEyeCenter);
            const rightOffset = getOffset(rightEyeCenter);

            leftPupil.style.transform = `translate(calc(-50% + ${leftOffset.x}px), calc(-50% + ${leftOffset.y}px))`;
            rightPupil.style.transform = `translate(calc(-50% + ${rightOffset.x}px), calc(-50% + ${rightOffset.y}px))`;
        }

        document.addEventListener('mousemove', movePupils);

        // Scroll animation for white content
        const whiteContent = document.getElementById('white-content');
        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    whiteContent.classList.add('visible');
                } else {
                    whiteContent.classList.remove('visible');
                }
            });
        }, { threshold: 0.3 });
        observer.observe(whiteContent);

        // Bubble repulsion effect (smooth)
        let bubbles = [];
        const container = document.querySelector('.white-content');
        const bubbleContainer = document.getElementById('bubbles');
        const repulsionRadius = 80;
        const repulsionStrength = 0.8;

        function createBubbles() {
            const bubbleCount = 20;
            for (let i = 0; i < bubbleCount; i++) {
                const bubble = document.createElement('div');
                bubble.className = 'bubble';

                const size = Math.random() * 40 + 10;
                bubble.style.width = size + 'px';
                bubble.style.height = size + 'px';

                const left = Math.random() * 100;
                const top = Math.random() * 100;
                bubble.style.left = left + '%';
                bubble.style.top = top + '%';

                bubble.dataset.origLeft = left;
                bubble.dataset.origTop = top;

                const duration = Math.random() * 6 + 4;
                bubble.style.animation = `float ${duration}s infinite ease-in-out`;
                bubble.style.animationDelay = Math.random() * 5 + 's';

                bubbleContainer.appendChild(bubble);
                bubbles.push(bubble);
            }
        }

        function handleMouseMove(e) {
            const containerRect = container.getBoundingClientRect();
            const mouseX = e.clientX - containerRect.left;
            const mouseY = e.clientY - containerRect.top;

            bubbles.forEach(bubble => {
                const bubbleRect = bubble.getBoundingClientRect();
                const bubbleX = bubbleRect.left + bubbleRect.width / 2 - containerRect.left;
                const bubbleY = bubbleRect.top + bubbleRect.height / 2 - containerRect.top;
                const dx = bubbleX - mouseX;
                const dy = bubbleY - mouseY;
                const dist = Math.sqrt(dx * dx + dy * dy);

                if (dist < repulsionRadius) {
                    const angle = Math.atan2(dy, dx);
                    const force = (repulsionRadius - dist) * repulsionStrength;
                    const offsetX = Math.cos(angle) * force;
                    const offsetY = Math.sin(angle) * force;

                    if (bubble.style.animation !== 'none') {
                        bubble.dataset.anim = bubble.style.animation;
                        bubble.style.animation = 'none';
                    }
                    bubble.style.transform = `translate(${offsetX}px, ${offsetY}px)`;
                } else {
                    if (bubble.style.animation === 'none' && bubble.dataset.anim) {
                        bubble.style.transform = '';
                        setTimeout(() => {
                            const currentDist = Math.sqrt(
                                (bubbleX - mouseX) ** 2 + (bubbleY - mouseY) ** 2
                            );
                            if (currentDist >= repulsionRadius) {
                                bubble.style.animation = bubble.dataset.anim || '';
                            }
                        }, 150);
                    }
                }
            });
        }

        let ticking = false;
        document.addEventListener('mousemove', (e) => {
            if (!ticking) {
                window.requestAnimationFrame(() => {
                    handleMouseMove(e);
                    ticking = false;
                });
                ticking = true;
            }
        });

        window.addEventListener('load', createBubbles);
    </script>
